Design goals

Universal scoring across different goals (workouts, minutes, weight loss/gain, calorie adherence days).

Outcome (did they move toward the goal at the rate they committed to?)

Balanced penalties: missing a week matters, but doesn’t crush motivation.

Quarterly seasons with soft reset:

Gold II resets around Silver II.

Platinum+ drops 2 tiers to keep top ranks rare.

13.2 Difficulty multipliers
function D_workouts(target: number): number {
  const table: Record<number, number> = {
    1: 0.70, 2: 0.85, 3: 1.00, 4: 1.18, 5: 1.40, 6: 1.70, 7: 1.85
  };
  return table[Math.round(target)] ?? 1.0;
}

function D_minutes(targetMinutes: number): number {
  const base = Math.pow(targetMinutes / 150, 0.55);
  return clamp(0.75, 2.0, base);
}

function D_calDays(targetDays: number): number {
  const table: Record<number, number> = {
    1: 0.75, 2: 0.82, 3: 0.90, 4: 0.98, 5: 1.08, 6: 1.22, 7: 1.40
  };
  return table[Math.round(targetDays)] ?? 1.0;
}

function D_weightLoss(params: {
  W0: number; Wg: number; Wt: number;
  Tweeks: number;
}): { D: number; D_base: number; lossTarget: number; progress: number } {
  const { W0, Wg, Wt } = params;
  const L = W0 - Wg;
  const Tweeks = Math.max(4, params.Tweeks);

  const p = clamp(0, 1, (W0 - Wt) / L);

  const D_base = 1 + 0.9 * Math.pow((L / W0), 0.6);
  const D_phase = 1 + 1.0 * Math.pow(p, 3.0);

  const lossTargetRaw = L / Tweeks;
  const lossTarget = clamp(0.25, 2.5, lossTargetRaw);

  const D_timeline = clamp(0.90, 1.60, Math.pow(lossTarget / 1.0, 0.35));

  const D = D_base * D_phase * D_timeline;
  return { D, D_base, lossTarget, progress: p };
}

function D_weightGain(params: {
  W0: number; Wg: number; Wt: number;
  Tweeks: number;
}): { D: number; D_base: number; gainTarget: number; progress: number } {
  const { W0, Wg, Wt } = params;
  const G = Wg - W0;
  const Tweeks = Math.max(4, params.Tweeks);

  const p = clamp(0, 1, (Wt - W0) / G);

  const D_base = 1 + 0.7 * Math.pow((G / W0), 0.6);
  const D_phase = 1 + 0.7 * Math.pow(p, 2.5);

  const gainTargetRaw = G / Tweeks;
  const gainTarget = clamp(0.10, 1.5, gainTargetRaw);

  const D_timeline = clamp(0.90, 1.50, Math.pow(gainTarget / 0.5, 0.35));

  const D = D_base * D_phase * D_timeline;
  return { D, D_base, gainTarget, progress: p };
}

13.4 GoalScore + WeekScore
const K = 100;
const wA = 0.70;
const wO = 0.30;

function goalScore(D: number, A: number, O: number): number {
  return K * D * (wA * A + wO * O);
}

function combineWeekScore(scores: number[]): number {
  if (scores.length === 0) return 0;
  const maxScore = Math.max(...scores);
  const avgScore = scores.reduce((a,b)=>a+b,0) / scores.length;
  return 0.60 * maxScore + 0.40 * avgScore;
}

function streakMultiplier(streakWeeks: number): number {
  if (streakWeeks >= 12) return 1.45;
  if (streakWeeks >= 8) return 1.30;
  if (streakWeeks >= 4) return 1.15;
  if (streakWeeks >= 2) return 1.05;
  return 1.00;
}

14.2 MMR → band & LP
function bandForMMR(mmr: number): Band {
  const x = Math.max(0, mmr);
  return BANDS.find(b => x >= b.min && x <= b.max) ?? BANDS[0];
}

function lpForMMR(mmr: number, band: Band): number {
  if (!band.division) return 0; // optional
  const lp = 100 * ((mmr - band.min) / (band.max - band.min));
  return clamp(0, 100, Math.round(lp));
}

15) Promotion/demotion evaluation with buffers + shield
Buffers

Division buffer: 40

Tier buffer: 120

const DIV_DEMOTE_BUFFER = 40;
const TIER_DEMOTE_BUFFER = 120;

Applying demotion logic (conceptual)

You will compute new MMR, then compute the “candidate” band.
But before allowing demotion, you check:

Is it a tier demotion?

Is tier shield active?

Pseudo:

function applyRankWithDemotionRules(params: {
  oldRank: Band;
  newMMR: number;
  tierShieldWeeksRemaining: number;
}): { finalBand: Band; lp: number } {

  const candidate = bandForMMR(params.newMMR);
  const oldBand = params.oldRank;

  // If candidate is same or higher, accept immediately (promotion is instant)
  if (isHigherOrEqual(candidate, oldBand)) {
    return { finalBand: candidate, lp: lpForMMR(params.newMMR, candidate) };
  }

  // Candidate is lower (demotion). Determine if it's tier demotion vs division demotion.
  const tierDemotion = candidate.tier !== oldBand.tier;

  if (tierDemotion && params.tierShieldWeeksRemaining > 0) {
    // Shield blocks tier demotion: keep at least old tier floor (but LP can be 0)
    const floored = floorToTier(oldBand.tier, params.newMMR);
    return { finalBand: floored, lp: lpForMMR(params.newMMR, floored) };
  }

  // Apply hysteresis: only demote if below boundary - buffer
  const threshold = demotionThreshold(oldBand);
  if (params.newMMR > threshold) {
    // Not low enough to demote; stay in old band (LP can decrease)
    return { finalBand: oldBand, lp: lpForMMR(params.newMMR, oldBand) };
  }

  // Demotion allowed
  return { finalBand: candidate, lp: lpForMMR(params.newMMR, candidate) };
}

// demotion threshold based on old band
function demotionThreshold(oldBand: Band): number {
  if (!oldBand.division) {
    // For Master/Challenger, you can decide special logic later; default:
    return oldBand.min - TIER_DEMOTE_BUFFER;
  }

  // Determine if demoting within tier or to lower tier by looking at band below
  const below = bandBelow(oldBand);
  const tierDemotion = below ? below.tier !== oldBand.tier : false;
  const buffer = tierDemotion ? TIER_DEMOTE_BUFFER : DIV_DEMOTE_BUFFER;
  return oldBand.min - buffer;
}
